Quick start

Docker Compose quick start

A trimmed quick-start version of the official Zammad Docker deployment — same images and services, comments and extras removed. Every secret is an .env placeholder.

docker-compose.ymlquick-start version
services:
  zammad-railsserver:
    image: ${IMAGE_REPO:-ghcr.io/zammad/zammad}:${VERSION:-7.2.3-0015}
    restart: ${RESTART:-always}
    command: ["zammad-railsserver"]
    environment:
      POSTGRESQL_HOST: ${POSTGRES_HOST:-zammad-postgresql}
      POSTGRESQL_DB: ${POSTGRES_DB:-zammad_production}
      POSTGRESQL_USER: ${POSTGRES_USER:-zammad}
      POSTGRESQL_PASS: ${POSTGRES_PASS:-zammad}
      REDIS_URL: ${REDIS_URL:-redis://zammad-redis:6379}
      MEMCACHE_SERVERS: ${MEMCACHE_SERVERS:-zammad-memcached:11211}
    volumes:
      - zammad-storage:/opt/zammad/storage
    depends_on:
      zammad-postgresql:
        condition: service_healthy
      zammad-redis:
        condition: service_healthy
      zammad-memcached:
        condition: service_healthy
  zammad-websocket:
    image: ${IMAGE_REPO:-ghcr.io/zammad/zammad}:${VERSION:-7.2.3-0015}
    restart: ${RESTART:-always}
    command: ["zammad-websocket"]
    environment:
      POSTGRESQL_HOST: ${POSTGRES_HOST:-zammad-postgresql}
      POSTGRESQL_DB: ${POSTGRES_DB:-zammad_production}
      POSTGRESQL_USER: ${POSTGRES_USER:-zammad}
      POSTGRESQL_PASS: ${POSTGRES_PASS:-zammad}
      REDIS_URL: ${REDIS_URL:-redis://zammad-redis:6379}
    volumes:
      - zammad-storage:/opt/zammad/storage
  zammad-scheduler:
    image: ${IMAGE_REPO:-ghcr.io/zammad/zammad}:${VERSION:-7.2.3-0015}
    restart: ${RESTART:-always}
    command: ["zammad-scheduler"]
    environment:
      POSTGRESQL_HOST: ${POSTGRES_HOST:-zammad-postgresql}
      POSTGRESQL_DB: ${POSTGRES_DB:-zammad_production}
      POSTGRESQL_USER: ${POSTGRES_USER:-zammad}
      POSTGRESQL_PASS: ${POSTGRES_PASS:-zammad}
      REDIS_URL: ${REDIS_URL:-redis://zammad-redis:6379}
    volumes:
      - zammad-storage:/opt/zammad/storage
  zammad-nginx:
    image: ${IMAGE_REPO:-ghcr.io/zammad/zammad}:${VERSION:-7.2.3-0015}
    restart: ${RESTART:-always}
    command: ["zammad-nginx"]
    ports:
      - "${NGINX_EXPOSE_PORT:-8080}:${NGINX_PORT:-8080}"
    environment:
      POSTGRESQL_HOST: ${POSTGRES_HOST:-zammad-postgresql}
      POSTGRESQL_DB: ${POSTGRES_DB:-zammad_production}
      POSTGRESQL_USER: ${POSTGRES_USER:-zammad}
      POSTGRESQL_PASS: ${POSTGRES_PASS:-zammad}
      REDIS_URL: ${REDIS_URL:-redis://zammad-redis:6379}
    volumes:
      - zammad-storage:/opt/zammad/storage
    depends_on:
      zammad-railsserver:
        condition: service_healthy
  zammad-postgresql:
    image: postgres:${POSTGRES_VERSION:-17.12-alpine}
    restart: ${RESTART:-always}
    environment:
      POSTGRES_USER: ${POSTGRES_SUPERUSER:-postgres}
      POSTGRES_PASSWORD: ${POSTGRES_PASS:-zammad}
      ZAMMAD_DB: ${POSTGRES_DB:-zammad_production}
      ZAMMAD_DB_USER: ${POSTGRES_USER:-zammad}
      ZAMMAD_DB_PASS: ${POSTGRES_PASS:-zammad}
    volumes:
      - postgresql-data:/var/lib/postgresql/data
  zammad-redis:
    image: redis:${REDIS_VERSION:-8.10.3-alpine}
    restart: ${RESTART:-always}
    volumes:
      - redis-data:/data
  zammad-memcached:
    image: memcached:${MEMCACHE_VERSION:-1.6.45-alpine}
    restart: ${RESTART:-always}
    command: memcached -m 256M
  zammad-elasticsearch:
    image: elasticsearch:${ELASTICSEARCH_VERSION:-9.5.5}
    restart: ${RESTART:-always}
    environment:
      discovery.type: single-node
      xpack.security.enabled: "false"
      ES_JAVA_OPTS: ${ELASTICSEARCH_JAVA_OPTS:--Xms1g -Xmx1g}
    volumes:
      - elasticsearch-data:/usr/share/elasticsearch/data
  zammad-init:
    image: ${IMAGE_REPO:-ghcr.io/zammad/zammad}:${VERSION:-7.2.3-0015}
    restart: on-failure
    command: ["zammad-init"]
    environment:
      POSTGRESQL_HOST: ${POSTGRES_HOST:-zammad-postgresql}
      POSTGRESQL_DB: ${POSTGRES_DB:-zammad_production}
      POSTGRESQL_USER: ${POSTGRES_USER:-zammad}
      POSTGRESQL_PASS: ${POSTGRES_PASS:-zammad}
      REDIS_URL: ${REDIS_URL:-redis://zammad-redis:6379}
    volumes:
      - zammad-storage:/opt/zammad/storage
  zammad-backup:
    image: ${IMAGE_REPO:-ghcr.io/zammad/zammad}:${VERSION:-7.2.3-0015}
    restart: ${RESTART:-always}
    command: ["zammad-backup"]
    environment:
      POSTGRESQL_HOST: ${POSTGRES_HOST:-zammad-postgresql}
      POSTGRESQL_DB: ${POSTGRES_DB:-zammad_production}
      POSTGRESQL_USER: ${POSTGRES_USER:-zammad}
      POSTGRESQL_PASS: ${POSTGRES_PASS:-zammad}
      REDIS_URL: ${REDIS_URL:-redis://zammad-redis:6379}
    volumes:
      - zammad-backup:/var/tmp/zammad
      - zammad-storage:/opt/zammad/storage
volumes:
  elasticsearch-data:
  postgresql-data:
  redis-data:
  zammad-backup:
  zammad-storage:

Save it as docker-compose.yml, put the ${...} values in a .env file next to it, then run docker compose up -d. The Self-host Starter Pack versions (20 apps) add per-app setup notes, backup pointers and hardening checks.

FAQ

Zammad VPS questions

Answers use only the checked dataset on this page — estimates are called out as estimates.

What are the minimum VPS requirements for Zammad?

In the RunOnVPS dataset, Zammad needs at least RAM 6 GB, CPU 2 vCPU, disk 20 GB. Some of these figures are estimated rather than officially published — the badge next to each value in the table above shows its basis. Our suggested starting tier, based on the recommended RAM, is the 16 GB RAM tier.

Does Zammad need a GPU?

No. Zammad does not need a GPU — the baseline deployment runs on CPU only.

What SaaS product does Zammad replace?

Zammad is catalogued here as a self-hosted alternative to Zendesk / Freshdesk. Self-hosting replaces the hosted subscription with your own VPS; feature parity varies, so compare your required workflow before migrating.
